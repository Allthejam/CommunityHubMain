            openBroadcastModal: function() {
                const modal = document.getElementById('broadcastModal');
                const title = document.getElementById('modalTitle');
                const body = document.getElementById('modalBody');
                const township = document.getElementById('modalTownship');

                const townshipName = this.getValue('township_name');
                township.innerText = `Jurisdiction: ${townshipName}`;

                let hub = this.getValue('primary_hub');
                if (this.failovers.hub) hub = `${this.getValue('secondary_hub')} [FAILOVER]`;

                let road = this.getValue('wf_escape_primary');
                if (this.failovers.route) road = `${this.getValue('wf_escape_secondary')} [SECONDARY BYPASS]`;

                if (this.currentEvent === 'wildfire') {
                    title.innerText = "WILDFIRE EMERGENCY EVACUATION";
                    body.innerText = `[LEVEL 3 EMERGENCY BROADCAST]\n\n"WILDFIRE ALERT: Active fire threatening ${townshipName}.\n\nPrimary Escape Route: ${road}.\nEvacuating residents proceed to Refuge Hub: ${hub}.\nCommand Post operational at: ${this.getValue('primary_hq')}."`;
                } else if (this.currentEvent === 'urbanfire') {
                    title.innerText = "URBAN STRUCTURAL FIRE CORDON";
                    body.innerText = `[LEVEL 3 EMERGENCY BROADCAST]\n\n"STRUCTURAL FIRE: Incident at ${this.getValue('uf_risk_blocks')}.\n\nSafety Cordon ${this.getValue('uf_cordon_dist')}m clear. Traffic divert via ${this.getValue('uf_bypass_route')}.\nWarmth assembly at ${this.getValue('uf_warmth_hub')}."`;
                } else if (this.currentEvent === 'flood') {
                    title.innerText = "SEPA FLOOD WARNING ALERT";
                    body.innerText = `[LEVEL 2 URGENT BROADCAST]\n\n"FLOOD WARNING: Heavy spate on ${this.getValue('fl_river')}.\n\nSandbag depot active at ${this.getValue('fl_sandbag_loc')} (Keyholder: ${this.getValue('fl_sandbag_tel')}).\nHigh Ground Refuge at ${this.getValue('fl_high_ground')}."`;
                } else if (this.currentEvent === 'power') {
                    title.innerText = "POWER OUTAGE WARM SPACE";
                    body.innerText = `[LEVEL 2 URGENT BROADCAST]\n\n"POWER OUTAGE EXCEEDS ${this.getValue('po_trigger_hrs')} HOURS.\n\nWarm Space active at ${hub} (${this.getValue('po_warm_hours')}).\nGenerator heating, canteen, and device charging operational."`;
                } else if (this.currentEvent === 'drought') {
                    title.innerText = "WATER SHORTAGE & BOTTLED WATER ALERT";
                    body.innerText = `[LEVEL 2 URGENT BROADCAST]\n\n"WATER SHORTAGE ALERT: Private Water Supplies (PWS) dried up for ${this.getValue('dr_pws_count')}.\n\nScottish Water bowser hardstanding active at ${this.getValue('dr_bowser_loc')}.\nBottled water distribution point open at ${this.getValue('dr_bottled_hub')} (10L/person/day).\n4x4 delivery runners dispatched for vulnerable households."`;
                } else if (this.currentEvent === 'unrest') {
                    title.innerText = "PUBLIC SAFETY NOTICE";
                    body.innerText = `[LEVEL 2 URGENT BROADCAST]\n\n"PUBLIC SAFETY ALERT: Avoid ${this.getValue('cu_avoid_area')}.\n\nResidents stay indoors. Official channel sync via Police Scotland Liaison ${this.getValue('cu_police_liaison')}."`;
                } else if (this.currentEvent === 'defence') {
                    title.innerText = "STATE EMERGENCY DISTRIBUTION";
                    body.innerText = `[LEVEL 3 EMERGENCY BROADCAST]\n\n"STATE EMERGENCY: Distribution Hub active at ${hub}.\n\nPotable water spring active at ${this.getValue('cd_water_spring')}.\nSubterranean shelter ready at ${this.getValue('cd_shelter_loc')}."`;
                }

                modal.classList.remove('hidden');
            },

                document.getElementById('broadcastModal').classList.add('hidden');
